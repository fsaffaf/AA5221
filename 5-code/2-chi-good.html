observed

barplot(observed,
main = "Political Party",
xlab = "Political Party",
ylab = "Frequency",
col = rainbow(length(observed)))

expected <- c(.40, .20, .40)

chi_result <- chisq.test(
x = observed,
p = expected
)

chi_result

w <- sqrt(

sum(observed)

Part 9: Calculate Cohen's W


w <- sqrt(
as.numeric(chi_result$statistic) /
sum(observed)
)

w
Cohen's W Interpretation
< 0.10 Negligible
0.10 - 0.29 Small
0.30 - 0.49 Moderate
≥ 0.50 Large
